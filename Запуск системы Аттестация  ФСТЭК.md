Сейчас система требует ручного запуска PostgreSQL и Ollama после каждой перезагрузки WSL2. Настроим автозапуск всего необходимого и создадим скрипт для быстрого старта.

---

## 1. Скрипт автозапуска

```bash
cat > ~/fstek-start.sh << 'STARTUP_EOF'
#!/bin/bash
# ============================================
# СКРИПТ ЗАПУСКА СИСТЕМЫ АТТЕСТАЦИИ ФСТЭК
# ============================================
# Использование:
#   bash ~/fstek-start.sh          — полный запуск (БД + UI)
#   bash ~/fstek-start.sh audit    — запуск аудита через CLI
#   bash ~/fstek-start.sh stop     — остановка всех сервисов
#   bash ~/fstek-start.sh status   — проверка статуса

set -e

PROJECT_DIR="$HOME/projects/fstek-attestation"
VENV_DIR="$PROJECT_DIR/.venv"

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

log_info()  { echo -e "${BLUE}[INFO]${NC} $1"; }
log_ok()    { echo -e "${GREEN}[OK]${NC} $1"; }
log_warn()  { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $1"; }

# ============================================
# Функции проверки и запуска компонентов
# ============================================

check_docker() {
    if docker ps > /dev/null 2>&1; then
        log_ok "Docker работает"
        return 0
    else
        log_warn "Docker не доступен, запускаю..."
        sudo service docker start 2>/dev/null || true
        sleep 3
        if docker ps > /dev/null 2>&1; then
            log_ok "Docker запущен"
            return 0
        else
            log_error "Не удалось запустить Docker"
            return 1
        fi
    fi
}

check_postgres() {
    if sudo -u postgres pg_isready -q 2>/dev/null; then
        log_ok "PostgreSQL работает (порт 5433)"
        return 0
    else
        log_warn "PostgreSQL не запущен, запускаю..."
        sudo pg_ctlcluster 16 main start 2>/dev/null || sudo service postgresql start 2>/dev/null || true
        sleep 3
        if sudo -u postgres pg_isready -q 2>/dev/null; then
            log_ok "PostgreSQL запущен"
            return 0
        else
            log_error "Не удалось запустить PostgreSQL"
            return 1
        fi
    fi
}

check_ollama() {
    if curl -s http://localhost:11434/api/tags > /dev/null 2>&1; then
        log_ok "Ollama работает (порт 11434)"
        return 0
    else
        log_warn "Ollama не запущен, запускаю..."
        ollama serve > /dev/null 2>&1 &
        sleep 5
        if curl -s http://localhost:11434/api/tags > /dev/null 2>&1; then
            log_ok "Ollama запущен"
            return 0
        else
            log_error "Не удалось запустить Ollama"
            return 1
        fi
    fi
}

check_chromadb() {
    if curl -s http://localhost:8001/api/v2/heartbeat > /dev/null 2>&1; then
        log_ok "ChromaDB работает (порт 8001)"
        return 0
    else
        log_warn "ChromaDB не запущен"
        return 1
    fi
}

check_redis() {
    if redis-cli -p 6380 ping > /dev/null 2>&1; then
        log_ok "Redis работает (порт 6380)"
        return 0
    else
        log_warn "Redis не запущен"
        return 1
    fi
}

start_containers() {
    log_info "Запуск Docker-контейнеров..."
    cd "$PROJECT_DIR"
    docker compose up -d 2>/dev/null || true
    sleep 5
    
    if check_chromadb && check_redis; then
        log_ok "Все контейнеры запущены"
    else
        log_warn "Некоторые контейнеры не запустились"
        docker compose ps
    fi
}

start_streamlit() {
    log_info "Запуск Streamlit UI..."
    cd "$PROJECT_DIR"
    source "$VENV_DIR/bin/activate"
    
    # Проверяем, не запущен ли уже
    if pgrep -f "streamlit run" > /dev/null; then
        log_warn "Streamlit уже запущен"
        return 0
    fi
    
    nohup streamlit run streamlit_app.py --server.port 8501 > /tmp/streamlit_fstek.log 2>&1 &
    sleep 5
    
    if pgrep -f "streamlit run" > /dev/null; then
        log_ok "Streamlit запущен: http://localhost:8501"
    else
        log_error "Streamlit не запустился. Лог: /tmp/streamlit_fstek.log"
    fi
}

stop_all() {
    log_info "Остановка всех сервисов..."
    
    # Остановка Streamlit
    pkill -f "streamlit run" 2>/dev/null && log_ok "Streamlit остановлен" || true
    
    # Остановка контейнеров
    cd "$PROJECT_DIR"
    docker compose down 2>/dev/null && log_ok "Контейнеры остановлены" || true
    
    # Остановка Ollama
    pkill -f "ollama serve" 2>/dev/null && log_ok "Ollama остановлен" || true
    
    log_info "Все сервисы остановлены"
}

show_status() {
    echo ""
    echo "============================================"
    echo "   СТАТУС СИСТЕМЫ АТТЕСТАЦИИ ФСТЭК"
    echo "============================================"
    
    # Docker
    docker ps > /dev/null 2>&1 && echo -e "${GREEN}✅${NC} Docker" || echo -e "${RED}❌${NC} Docker"
    
    # PostgreSQL
    sudo -u postgres pg_isready -q 2>/dev/null && echo -e "${GREEN}✅${NC} PostgreSQL (5433)" || echo -e "${RED}❌${NC} PostgreSQL"
    
    # ChromaDB
    curl -s http://localhost:8001/api/v2/heartbeat > /dev/null 2>&1 && echo -e "${GREEN}✅${NC} ChromaDB (8001)" || echo -e "${RED}❌${NC} ChromaDB"
    
    # Redis
    redis-cli -p 6380 ping > /dev/null 2>&1 && echo -e "${GREEN}✅${NC} Redis (6380)" || echo -e "${RED}❌${NC} Redis"
    
    # Ollama
    curl -s http://localhost:11434/api/tags > /dev/null 2>&1 && echo -e "${GREEN}✅${NC} Ollama (11434)" || echo -e "${RED}❌${NC} Ollama"
    
    # Streamlit
    pgrep -f "streamlit run" > /dev/null && echo -e "${GREEN}✅${NC} Streamlit (8501)" || echo -e "${YELLOW}⚠️${NC}  Streamlit (не запущен)"
    
    # GPU
    nvidia-smi > /dev/null 2>&1 && echo -e "${GREEN}✅${NC} GPU доступна" || echo -e "${RED}❌${NC} GPU"
    
    echo "============================================"
    echo ""
    
    # Показать URL если Streamlit запущен
    if pgrep -f "streamlit run" > /dev/null; then
        echo -e "${GREEN}🌐 Streamlit: http://localhost:8501${NC}"
    fi
}

# ============================================
# Главная логика
# ============================================

case "${1:-start}" in
    start|full)
        echo ""
        echo "============================================"
        echo "   ЗАПУСК СИСТЕМЫ АТТЕСТАЦИИ ФСТЭК"
        echo "============================================"
        echo ""
        
        check_docker
        check_postgres
        start_containers
        check_ollama
        
        echo ""
        log_info "Все сервисы запущены!"
        echo ""
        echo -e "Для запуска интерфейса: ${GREEN}bash ~/fstek-start.sh ui${NC}"
        echo -e "Для запуска аудита:   ${GREEN}bash ~/fstek-start.sh audit${NC}"
        echo ""
        ;;
    
    ui)
        check_docker
        check_postgres
        start_containers
        check_ollama
        start_streamlit
        echo ""
        echo -e "${GREEN}🌐 Streamlit запущен: http://localhost:8501${NC}"
        echo -e "Лог: /tmp/streamlit_fstek.log"
        echo -e "Для остановки: ${YELLOW}bash ~/fstek-start.sh stop${NC}"
        echo ""
        ;;
    
    audit)
        check_docker
        check_postgres
        start_containers
        check_ollama
        
        log_info "Запуск аудита..."
        cd "$PROJECT_DIR"
        source "$VENV_DIR/bin/activate"
        
        if [ -n "$2" ]; then
            python run_audit.py "$2"
        else
            python run_audit.py input_data.json
        fi
        ;;
    
    stop)
        stop_all
        ;;
    
    status)
        show_status
        ;;
    
    restart)
        stop_all
        sleep 3
        bash "$0" start
        ;;
    
    *)
        echo "Использование: bash ~/fstek-start.sh [команда]"
        echo ""
        echo "Команды:"
        echo "  start    — запуск всех сервисов (по умолчанию)"
        echo "  ui       — запуск сервисов + Streamlit UI"
        echo "  audit    — запуск сервисов + аудит"
        echo "  stop     — остановка всех сервисов"
        echo "  status   — проверка статуса"
        echo "  restart  — перезапуск"
        ;;
esac
STARTUP_EOF

chmod +x ~/fstek-start.sh
```

## 2. Настройка автозапуска PostgreSQL в WSL2

```bash
# Создаём задачу для автозапуска PostgreSQL при входе в WSL2
echo 'sudo pg_ctlcluster 16 main start 2>/dev/null || sudo service postgresql start 2>/dev/null || true' >> ~/.bashrc
```

## 3. Инструкция: запуск системы после перезагрузки

```bash
cat > ~/FSTEK_README.md << 'README_EOF'
# Инструкция по запуску системы аттестации ФСТЭК

## После перезагрузки компьютера

### 1. Запустить Docker Desktop (Windows)

Нажмите Пуск → Docker Desktop. Дождитесь зелёного индикатора в трее.

### 2. Открыть WSL2

Запустите терминал Ubuntu 24.04.

### 3. Запустить систему

```bash
# Полный запуск всех сервисов
bash ~/fstek-start.sh

# Запуск сервисов + Streamlit UI
bash ~/fstek-start.sh ui

# Проверка статуса
bash ~/fstek-start.sh status
```

### 4. Открыть интерфейс

Браузер → http://localhost:8501
